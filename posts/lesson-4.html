ملخص: أمر ALTER TABLE في MySQL

يُستخدم الأمر لتعديل هيكل جدول موجود: إضافة الأعمدة وحذفها وتعديلها، وإدارة القيود (UNIQUE، CHECK، DEFAULT، PRIMARY KEY) دون المساس بالبيانات المخزنة.

حمّل ملخص المحاضرة

ملف PDF منسّق وجاهز للمراجعة والطباعة دون اتصال بالإنترنت

1

الصيغة العامة وإضافة وحذف الأعمدة

الصيغة العامة للأمر:

ALTER TABLE table_name action;

إضافة عمود

ALTER TABLE Employees ADD COLUMN Salary DECIMAL(10,2);

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ADD Salary DECIMAL(10,2);

حذف عمود

ALTER TABLE Employees DROP COLUMN Salary;

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees DROP Salary;
💡

ملاحظة: كلمة COLUMN اختيارية، يمكنك كتابتها أو تجاهلها.

2

تغيير اسم الجدول

ALTER TABLE Employees RENAME TO Staff;
3

تعديل نوع البيانات وقيد NOT NULL (MODIFY)

تُستخدم MODIFY لتعديل نوع بيانات العمود، أو إضافة/إزالة قيد NOT NULL، بدون تغيير اسم العمود.

التعديل وإضافة القيد

ALTER TABLE Employees MODIFY Age INT NOT NULL;

إزالة قيد NOT NULL (السماح بالقيم الفارغة NULL)

ALTER TABLE Employees MODIFY Age INT NULL;
4

تعديل اسم العمود أو نوعه (CHANGE)

تُستخدم CHANGE لتغيير اسم العمود أو نوع البيانات أو كليهما معاً، وتتطلب كتابة الاسم القديم ثم الجديد. 

ALTER TABLE Employees CHANGE Age Employee_Age INT;
5

قيد UNIQUE (فريد)

الإضافة بدون تحديد اسم (يسميه النظام تلقائياً)

ALTER TABLE Employees ADD UNIQUE (Email);

الإضافة مع اسم مخصص (CONSTRAINT)

ALTER TABLE Employees ADD CONSTRAINT UQ_Email UNIQUE (Email);

حذف القيد (في إصدارات MySQL الحديثة)

ALTER TABLE Employees DROP CONSTRAINT UQ_Email;
6

قيد CHECK (التحقق من شرط)

الإضافة بدون تحديد اسم

ALTER TABLE Employees ADD CHECK (Age >= 18);

الإضافة مع اسم مخصص

ALTER TABLE Employees ADD CONSTRAINT CHK_Age CHECK (Age >= 18);

حذف القيد

ALTER TABLE Employees DROP CONSTRAINT CHK_Age;
7

قيد DEFAULT (القيمة الافتراضية)

💡

ملاحظة: كلمة COLUMN بعد ALTER الثانية اختيارية، ويمكنك كتابتها أو حذفها.

إضافة القيد

ALTER TABLE Employees ALTER COLUMN City SET DEFAULT 'Amman';

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ALTER City SET DEFAULT 'Amman';

حذف القيد

ALTER TABLE Employees ALTER COLUMN City DROP DEFAULT;

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ALTER City DROP DEFAULT;
8

الاستعلام عن القيود في الجدول

لمعرفة القيود التي أضفتها على جدول معين، استخدم هذا الأمر:

SELECT CONSTRAINT_NAME, CONSTRAINT_TYPE
FROM information_schema.TABLE_CONSTRAINTS
WHERE TABLE_NAME = 'Employees'
  AND TABLE_SCHEMA = 'اسم_قاعدة_البيانات_هنا';

شرح الأمر:

اسحب الجدول جانبياً لعرض باقي الأعمدة

شرح أجزاء أمر الاستعلام عن القيود
الجزء المعنى
information_schema.TABLE_CONSTRAINTS «دليل النظام» داخل الخادم، ويحتوي على بيانات جميع القيود (Metadata).
CONSTRAINT_NAME يعرض اسم القيد، سواء كان مخصصاً مثل UQ_Email أو تلقائياً.
CONSTRAINT_TYPE يعرض نوع القيد (UNIQUE، CHECK، إلخ).
WHERE TABLE_NAME = 'Employees' يفلتر النتائج لتعرض القيود الخاصة بجدول الموظفين فقط.
AND TABLE_SCHEMA = '...' يحدد اسم قاعدة البيانات.
⚠️

شرط في غاية الأهمية: بدون TABLE_SCHEMA، إذا كان لديك جدول باسم Employees في قواعد بيانات أخرى على نفس الخادم (مثل قاعدة اختبار وقاعدة إنتاج) فستظهر قيودها جميعاً وتختلط النتائج. تحديده يضمن الحصول على قيود الجدول في القاعدة الحالية فقط.

9

المفتاح الأساسي (PRIMARY KEY)

إضافة القيد (للعمود الموجود بالفعل)

ALTER TABLE Employees ADD PRIMARY KEY (ID);
💡

لجعل المفتاح الأساسي مركباً من أكثر من عمود (Composite Key) افصل بين الأعمدة بفاصلة:

ADD PRIMARY KEY (ID, Department_ID)

إضافة القيد مع كتابة اسم (CONSTRAINT)

ALTER TABLE Employees ADD CONSTRAINT PK_Employee PRIMARY KEY (ID);
💡

معلومة خاصة بـ MySQL: حتى لو كتبت اسماً مخصصاً مثل PK_Employee، يتجاهله النظام ويحفظ المفتاح الأساسي دائماً باسم PRIMARY.

حذف القيد

لا يمكن أن يحتوي الجدول على أكثر من مفتاح أساسي واحد، لذلك لا تحتاج إلى تحديد اسمه عند الحذف:

ALTER TABLE Employees DROP PRIMARY KEY;
⚠️

ملاحظة هامة قبل الحذف: إذا كان عمود المفتاح الأساسي يحتوي على خاصية الترقيم التلقائي (AUTO_INCREMENT)، فيجب أولاً إزالة هذه الخاصية باستخدام MODIFY أو CHANGE، وإلا سيرفض النظام حذف المفتاح الأساسي.
