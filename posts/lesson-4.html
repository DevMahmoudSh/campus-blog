أوامر ALTER TABLE في MySQL

دليل تطبيقي لأوامر ALTER في MySQL: الصيغة العامة، وإدارة الأعمدة والقيود، والاستعلام عن القيود، مع أمثلة جاهزة للنسخ.

حمّل ملخص المحاضرة

الصيغة العامة لأمر ALTER في MySQL:

ALTER TABLE table_name action;
1

إضافة وحذف عمود

إضافة عمود

تعتبر كلمة COLUMN اختيارية، يمكنك كتابتها أو تجاهلها. 


ALTER TABLE Employees ADD COLUMN Salary DECIMAL(10,2);

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ADD Salary DECIMAL(10,2);

حذف عمود


ALTER TABLE Employees DROP COLUMN Salary;

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees DROP Salary;
2

تغيير اسم الجدول

ALTER TABLE Employees RENAME TO Staff;
3

تعديل نوع البيانات وإضافة قيد NOT NULL (باستخدام MODIFY)

تُستخدم MODIFY لتعديل نوع بيانات العمود أو إضافة/إزالة قيد NOT NULL بدون تغيير اسم العمود.

التعديل وإضافة القيد

ALTER TABLE Employees MODIFY Age INT NOT NULL;

إزالة قيد NOT NULL (إرجاعه ليقبل القيم الفارغة NULL)

ALTER TABLE Employees MODIFY Age INT NULL;
4

تعديل اسم العمود أو نوعه (باستخدام CHANGE)

تُستخدم CHANGE لتغيير اسم العمود، أو نوع البيانات، أو كليهما معاً (تتطلب كتابة الاسم القديم ثم الجديد). 

ALTER TABLE Employees CHANGE Age Employee_Age INT;
5

قيد UNIQUE (فريد)

الإضافة (بدون تحديد اسم للقيد - يقوم النظام بتسميته تلقائياً)

ALTER TABLE Employees ADD UNIQUE (Email);

الإضافة (مع تحديد اسم مخصص للقيد باستخدام CONSTRAINT)

ALTER TABLE Employees ADD CONSTRAINT UQ_Email UNIQUE (Email);

حذف قيد UNIQUE (في إصدارات MySQL الحديثة)

ALTER TABLE Employees DROP CONSTRAINT UQ_Email;
6

قيد CHECK (التحقق من شرط)

الإضافة (بدون تحديد اسم للقيد)

ALTER TABLE Employees ADD CHECK (Age >= 18);

الإضافة (مع تحديد اسم مخصص للقيد)

ALTER TABLE Employees ADD CONSTRAINT CHK_Age CHECK (Age >= 18);

حذف قيد CHECK

ALTER TABLE Employees DROP CONSTRAINT CHK_Age;
7

قيد DEFAULT (القيمة الافتراضية)

💡

ملاحظة: كلمة COLUMN بعد ALTER الثانية هي اختيارية، ويمكنك كتابتها أو حذفها.

إضافة القيد


ALTER TABLE Employees ALTER COLUMN City SET DEFAULT 'Amman';

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ALTER City SET DEFAULT 'Amman';

حذف قيد DEFAULT


ALTER TABLE Employees ALTER COLUMN City DROP DEFAULT;

-- أو بدون كتابة كلمة COLUMN
ALTER TABLE Employees ALTER City DROP DEFAULT;
8

الاستعلام عن القيود في الجدول (قراءة هيكل القيود)

لمعرفة القيود التي قمت بإضافتها على جدول معين، استخدم هذا الأمر: 


SELECT CONSTRAINT_NAME, CONSTRAINT_TYPE
FROM information_schema.TABLE_CONSTRAINTS
WHERE TABLE_NAME = 'Employees'
  AND TABLE_SCHEMA = 'اسم_قاعدة_البيانات_هنا';

شرح الأمر:

الجزء الشرح
information_schema.TABLE_CONSTRAINTS هو "دليل نظام" داخل الخادم يحتوي على بيانات لجميع القيود (Metadata).
CONSTRAINT_NAME يعرض اسم القيد (سواء الاسم المخصص مثل UQ_Email أو التلقائي).
CONSTRAINT_TYPE يعرض نوع القيد (UNIQUE, CHECK، إلخ).
WHERE TABLE_NAME = 'Employees' يفلتر النتائج ليعرض فقط القيود الخاصة بجدول الموظفين.
AND TABLE_SCHEMA = '...' شرط في غاية الأهمية؛ فهو يحدد "اسم قاعدة البيانات".
⚠️

لماذا شرط TABLE_SCHEMA مهم؟ بدون هذا الشرط، إذا كان لديك جدول باسم Employees في قواعد بيانات أخرى على نفس الخادم (مثل قاعدة اختبار وقاعدة إنتاج)، ستظهر لك قيودها جميعاً وتختلط النتائج. تحديده يضمن الحصول على قيود الجدول في القاعدة الحالية فقط.

9

إضافة وحذف المفتاح الأساسي (PRIMARY KEY)

تتم إضافة وحذف المفتاح الأساسي باستخدام ALTER TABLE في MySQL بالصيغ التالية:

إضافة قيد PRIMARY KEY

إذا كان العمود موجوداً بالفعل وتريد تعيينه كمفتاح أساسي: 

ALTER TABLE Employees ADD PRIMARY KEY (ID);
💡

ملاحظة: إذا أردت جعل المفتاح الأساسي مركباً من أكثر من عمود (Composite Key)، تفصل بينها بفاصلة: ADD PRIMARY KEY (ID, Department_ID)

إضافة قيد PRIMARY KEY مع كتابة اسم للقيد (CONSTRAINT)

ALTER TABLE Employees ADD CONSTRAINT PK_Employee PRIMARY KEY (ID);
💡

معلومة خاصة بـ MySQL: حتى إذا قمت بكتابة اسم مخصص للمفتاح الأساسي مثل PK_Employee، فإن النظام سيتجاهل هذا الاسم ويحفظه دائماً باسم PRIMARY في الكواليس.

حذف قيد PRIMARY KEY

نظراً لأن أي جدول لا يمكنه احتواء أكثر من مفتاح أساسي واحد، فإنك لا تحتاج إلى تحديد اسمه عند الحذف، بل يكفي كتابة الأمر التالي مباشرة: 

ALTER TABLE Employees DROP PRIMARY KEY;
⚠️

ملاحظة هامة قبل الحذف: إذا كان عمود المفتاح الأساسي يحتوي على خاصية الترقيم التلقائي (AUTO_INCREMENT)، يجب عليك أولاً إزالة هذه الخاصية باستخدام MODIFY أو CHANGE، وإلا سيرفض النظام حذف المفتاح الأساسي.
