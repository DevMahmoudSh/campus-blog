ملخص شامل ومنظم لدرس DDL CREATE

في هذا الدرس سنغوص في أعماق بناء قواعد البيانات، بدءاً من ضبط الترميز الصحيح، مروراً بهندسة الأعمدة والقيود، وصولاً إلى أوامر الإفراغ والهدم.

1

ضبط بيئة التخزين والترميز

قاعدة البيانات تمثل "مساحة التخزين الفعلية" (الدفتر)، والتطبيق يمثل "لغة النقل" (ساعي البريد). لضمان استيعاب كافة اللغات والحروف والرموز التعبيرية بدون مشاكل الترميز (???? أو تشوه النصوص):

يدعم تخزين جميع لغات العالم مع الرموز التعبيرية (Emojis) بدقة تصل إلى 4 بايت لكل رمز.
utf8mb4 Character Set
يحدد قواعد الترتيب والمقارنة المعتمدة عالمياً مع تجاهل حساسية حالة الأحرف الكبيرة والصغيرة (Case-Insensitive).
utf8mb4_unicode_ci Collation 

-- إنشاء قاعدة البيانات مع تحديد الترميز وقواعد الترتيب صراحةً
CREATE DATABASE IF NOT EXISTS UniversityDB
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE UniversityDB;
2

الهيكلية المعيارية لبناء العمود

عند تعريف أي عمود داخل الجدول، يتبع محرك قواعد البيانات ترتيباً منطقياً محدداً:

القيد / الخاصية الدور الهندسي
NOT NULL يمنع ترك الحقل فارغاً، ويُستخدم للحقول الإلزامية في النظام.
DEFAULT يُسند قيمة تلقائية للحقل في حال لم يرسلها التطبيق أثناء الإضافة.
AUTO_INCREMENT يُولّد أرقاماً تسلسلية فريدة تلقائياً مع كل صف جديد.
UNIQUE يمنع تكرار القيمة في العمود عبر كامل الجدول (مثل البريد ورقم الهوية).
CHECK يتحقق منطقياً من مطابقة البيانات لشرط معين قبل قبولها.
PRIMARY KEY المعرّف الفريد للجدول (يجمع تلقائياً بين NOT NULL و UNIQUE).
FOREIGN KEY يضمن التكامل المرجعي بربط العمود بمفتاح أساسي في جدول آخر.
3

القيود: مستوى العمود مقابل الجدول

  • على مستوى العمود (Column-level): يُكتب بجانب اسم العمود مباشرة، ويُستخدم عندما يخص القيد عموداً واحداً فقط. (ملاحظة: NOT NULL و DEFAULT لا يُكتبان إلا هنا).
  • على مستوى الجدول (Table-level): يُكتب في نهاية الجدول، وتكمن أهميته في:
    • إمكانية تسمية القيد (CONSTRAINT constraint_name) لتسهيل إدارته لاحقاً.
    • إنشاء المفاتيح المركبة (Composite Keys) التي تجمع أكثر من عمود.

الكود التطبيقي للدمج بين المستويين:


-- أولاً: إنشاء الجدول الأب (Departments)
CREATE TABLE Departments (
    DeptID INT AUTO_INCREMENT PRIMARY KEY,
    DeptName VARCHAR(100) NOT NULL UNIQUE
);

-- ثانياً: إنشاء الجدول التابع (Employees) موضحاً المستويين
CREATE TABLE Employees (
    -- قيود على مستوى العمود (Column-level)
    EmployeeID INT AUTO_INCREMENT PRIMARY KEY,
    FullName VARCHAR(100) NOT NULL,
    Email VARCHAR(150) NOT NULL UNIQUE,
    AccountStatus VARCHAR(20) NOT NULL DEFAULT 'Active',
    Salary DECIMAL(10, 2) NOT NULL,
    DeptID INT,

    -- قيود على مستوى الجدول (Table-level)
    -- 1. قيد تحقق منطقي مسمى (Named Check Constraint)
    CONSTRAINT chk_minimum_salary CHECK (Salary >= 3000.00),

    -- 2. مفتاح أجنبي يربط الموظف بقسمه مع تحديد سلوك الحذف
    CONSTRAINT fk_employee_dept FOREIGN KEY (DeptID) 
        REFERENCES Departments(DeptID) 
        ON DELETE SET NULL
);

مثال المفتاح الأساسي المركب (Composite Primary Key):


-- جدول تسجيل المواد: لا يمكن للطالب تسجيل نفس المادة مرتين
CREATE TABLE CourseEnrollments (
    StudentID INT NOT NULL,
    CourseID INT NOT NULL,
    EnrollmentDate DATE DEFAULT (CURRENT_DATE),

    -- مفتاح أساسي مركب يتكون من عمودين معاً
    CONSTRAINT pk_student_course PRIMARY KEY (StudentID, CourseID)
);
4

أوامر الإفراغ والهدم (TRUNCATE مقابل DROP)

كلا الأمرين يندرجان تحت لغة تعريف البيانات (DDL)، ولكل منهما سلوك تشغيلي مختلف تماماً:

TRUNCATE TABLE

يُفرغ الجدول من جميع البيانات فوراً ويُعيد تصفير عداد AUTO_INCREMENT إلى قيمته الابتدائية (1)، مع الاحتفاظ بهيكل الجدول والأعمدة والقيود كاملة.

DROP TABLE

يمحو الجدول بالكامل من القرص الصلب (البيانات والهيكل والقيود معاً)، ولا يمكن التراجع عن هذا الإجراء نهائياً. 


-- إفراغ جدول التسجيل من السجلات مع بقاء الجدول جاهزاً للاستخدام
TRUNCATE TABLE CourseEnrollments;

-- حذف الجداول بالكامل وإزالتها من الوجود
DROP TABLE IF EXISTS CourseEnrollments;
DROP TABLE IF EXISTS Employees;
DROP TABLE IF EXISTS Departments;

-- حذف قاعدة البيانات بالكامل
DROP DATABASE IF EXISTS UniversityDB;
